for each parallel region R {

    State *S = state_of(R);

    if (S->first_execution) {

        /* cache/TLB warm-up */
        run(R, MAX_THREADS, SCATTER);
        S->phase = TM_SEARCH;
        continue;
    }


    /* ==============================================
     * 1. Thread mapping search
     * ============================================== */

    if (S->phase == TM_SEARCH) {

        for TM in {
            // CONTIGUOUS, 没有 SMT的机器上 这个没有意义
            SCATTER,
            CLOSE
        } {
            run_and_measure_XObject(
                R,
                TM,
                FIRST_TOUCH,
                MAX_THREADS
            );
        }

        S->best_TM = lowest_XObject_TM();


        /* ==========================================
         * 2. Page mapping decision
         * ========================================== */

        // 暂时不实现 方一个空函数 TODO
        // if (stddev(XObject) > 10%)     
        //     enable_linux_NUMA_balancing(); // 放一个空函数 暂时不需要实现


        S->phase = NT_SEARCH;
    }


    /* ==============================================
     * 3. Thread-count search
     * ============================================== */

    if (S->phase == NT_SEARCH) {

        /*
         * search space:
         *
         * {2, 4, 6, ..., MAX_THREADS}
         * 
         * Fibonacci search avoids exhaustive search.
         */
        nt = fibonacci_search_next();

        run_and_measure_XObject(
            R,
            S->best_TM,
            current_PM,
            nt
        );

        update_fibonacci_search();

        if (search_converged()) {
            S->best_NT = best_nt();
            S->phase = STABLE;
        }
    }


    /* ==============================================
     * 4. Stable execution
     * ============================================== */

        //         t,

    }
}
